valid = {

Detección de outliers (IQR)

Se marcan outliers con la regla 1.5·IQR en 5 variables; no se eliminan, solo se señalan con columnas outlier_*.

>_ Python · código
# 7. Detectar outliers mediante IQR
outlier_cols = ["edad","pres_art_reposo","colesterol","frec_cardica","depresion"]
for col in outlier_cols:
    q1 = df[col].quantile(0.25)
    q3 = df[col].quantile(0.75)
    iqr = q3 - q1
    limite_inf = q1 - 1.5 * iqr
    limite_sup = q3 + 1.5 * iqr
    df[f"outlier_{col}"] = (df[col] < limite_inf) | (df[col] > limite_sup)
